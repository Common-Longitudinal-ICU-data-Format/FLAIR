\begin{frame}{ICU AI has a trust bottleneck}
  \begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \begin{itemize}
      \item Models for sepsis, extubation, mortality, and operations now appear \emphc{faster than evidence can validate them}
      \item A strong internal AUC does not answer the question a hospital actually has
      \item The hard part is not training another model; it is proving whether a model works \emphc{somewhere it was not built}
    \end{itemize}
  \end{column}
  \begin{column}{0.45\textwidth}
    \begin{block}{The clinical-scientific question}
      Before we trust or deploy this model, can it survive contact with our ICU population, our workflows, and our data?
    \end{block}
  \end{column}
  \end{columns}
  \vfill
  \bigclaim{FLAIR turns that question into a repeatable network test.}
\end{frame}

  \vfill
  \begin{center}

    {\Large Keep patient-level data inside each institution.}

  \vfill
  \begin{center}\color{flairblue}\Large\textbf{The model goes to the data. Only aggregate, cell-suppressed reports come back.}\end{center}

\end{frame}

\section{Why FLAIR}

\begin{frame}{Public benchmarks are necessary, but not sufficient}
  \begin{center}
  \tiny
  \setlength{\tabcolsep}{4pt}
  \rowcolors{2}{flairpanel}{white}
  \begin{tabular}{>{\raggedright\arraybackslash}p{2.35cm}>{\raggedright\arraybackslash}p{4.15cm}>{\raggedright\arraybackslash}p{4.15cm}}
  \toprule
  \textbf{Dimension} & \textbf{Public ICU benchmarks} & \textbf{FLAIR} \\
  \midrule
  Test data & centralized / single-site & \emphc{federated}; stays at each hospital \\
  Main value & reproducible development & \emphc{external validation} under real institutional variation \\
  Tasks & mortality, LOS, AKI, generic sepsis & deployment-facing mortality, LTACH, extubation, CDC-ASE sepsis, readmission \\
  Split & often global or random & \emphc{temporal} at external sites; pseudo-prospective \\
  Scoring & frequently row-level & episodic / landmark / peak; closer to operational use \\
  Saturation risk & public test data can be tuned against & private site data never enter public hands \\
  \bottomrule
  \end{tabular}
  \end{center}
  \vfill
  {\color{flairmuted}\scriptsize FLAIR complements MIMIC, eICU, HiRID, YAIB, EHRSHOT, and MEDS. It adds the missing federated validation layer.}
  \source{van de Water et al. 2023; Wornow et al. 2023; Arnrich et al. 2025}
\end{frame}

\begin{frame}{External validation is too slow today}
  \begin{columns}[T]
  \begin{column}{0.52\textwidth}
    \begin{itemize}
      \item Independent EHR model validation usually requires one-off data-use agreements, local analyst time, and repeated reimplementation
      \item Flagship systems have taken \emphc{years} to reach only a handful of sites
      \item The Epic Sepsis Model fell to \rushc{AUC 0.63} in independent evaluation, against vendor-claimed 0.76--0.83
    \end{itemize}
  \end{column}
  \begin{column}{0.44\textwidth}
    \begin{block}{FLAIR changes the unit of work}
      From bespoke bilateral studies to a standing, governed model-to-data protocol over a shared ICU standard.
    \end{block}
  \end{column}
  \end{columns}
  \source{Wong et al., JAMA Intern Med 2021; Habib et al. 2021; Sendak et al. 2025; Adams et al. 2022}
\end{frame}

\begin{frame}{The infrastructure gap: no standing ICU validation network}
  \begin{itemize}
    \item \emphc{OHDSI / LEGEND} federates studies over OMOP, but not an ICU benchmark catalog
    \item \emphc{OpenSAFELY} moves analytic code to secure data, but not critical-care model benchmarking
    \item \emphc{MedPerf} and \emphc{FeTS} prove model-to-data and federated training in imaging
    \item \emphc{DREAM / EHR DREAM} show challenge-style model-to-data evaluation
  \end{itemize}
  \vfill
  \bigclaim{Missing for ICU EHR: CLIF standardization + reusable, extensible tasks + governed federated evaluation.}
  \source{Reps et al. 2020; OpenSAFELY 2021; Karargyris et al. 2023; Pati et al. 2022; Guinney et al. 2018; Bergquist et al. 2024}
\end{frame}

\section{What FLAIR Is}

\begin{frame}{FLAIR is three coupled products}
  \begin{columns}[T]
  \begin{column}{0.32\textwidth}
    \begin{block}{1. Data layer}
      CLIF critical-care standard plus public MIMIC-CLIF entry point for development.
    \end{block}
  \end{column}
  \begin{column}{0.32\textwidth}
    \begin{block}{2. Task layer}
      Five inspectable Python tasks today; open, governed, PR-extensible catalog.
    \end{block}
  \end{column}
  \begin{column}{0.32\textwidth}
    \begin{block}{3. Protocol layer}
      Committee-reviewed model-to-data execution with aggregate-only reports.
    \end{block}
  \end{column}
  \end{columns}
  \vfill
  \begin{center}\large\color{flairblue}\textbf{This is not another ICU model. It is the evidence layer models must pass through.}\end{center}
\end{frame}

\begin{frame}{FLAIR is open infrastructure for ICU evaluation standards}
  \begin{columns}[T]
  \begin{column}{0.53\textwidth}
    \begin{itemize}
      \item Open-source task definitions, feature contracts, report generation, and evaluation protocol
      \item A common way to define cohorts, prediction times, labels, splits, metrics, report bundles, and audit metadata
      \item New tasks are added as inspectable Python modules, not one-off private benchmark forks
    \end{itemize}
  \end{column}
  \begin{column}{0.43\textwidth}
    \begin{block}{Task catalog standard}
      Contribution requires MIMIC-CLIF reproducibility, clinician face-validity review, and multi-site governance sign-off.
    \end{block}
  \end{column}
  \end{columns}
  \vfill
  \begin{center}\color{flairblue}\textbf{The goal is reusable, auditable, extensible ICU AI evaluation across the CLIF community.}\end{center}
\end{frame}

\begin{frame}{What each stakeholder gets}
  \begin{center}
  \small
  \setlength{\tabcolsep}{5pt}
  \rowcolors{2}{flairpanel}{white}
  \begin{tabular}{>{\raggedright\arraybackslash}p{3.2cm}>{\raggedright\arraybackslash}p{8.1cm}}
  \toprule
  \textbf{Stakeholder} & \textbf{FLAIR value} \\
  \midrule
  Model developers & A public development path plus credible external evidence without acquiring private ICU data \\
  Hospitals & Local control: approve what runs, keep patient rows inside the institution, receive comparable performance reports \\
  Journals / reviewers & Audit hashes, reproducible task code, cohort summaries, confidence intervals, calibration where meaningful \\
  CLIF consortium & A high-value use case that makes CLIF adoption immediately useful for benchmarking and collaboration \\
  Patients / public & Less reliance on claims from one development hospital before real-world evaluation \\
  \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

\begin{frame}{Why CLIF makes the network possible}
  \begin{itemize}
    \item CLIF maps ICU EHR content into shared relational tables and mCIDE categories: vitals, labs, medications, respiratory support, microbiology, ADT, diagnoses, and more
    \item Task code refers to \emphc{shared table names and category values}, not local idiosyncratic EHR extracts
    \item MIMIC-CLIF expresses MIMIC-IV in CLIF v2.1 with approximately \emphc{72{,}800 ICU stays}
    \item Result: developers debug publicly, then run the same task/model contract across private CLIF sites
  \end{itemize}
  \vfill
  \begin{center}\color{flairblue}\textbf{CLIF is the portability layer. FLAIR is the validation layer.}\end{center}
  \source{Rojas et al. 2025; Johnson et al. 2023}
\end{frame}

\begin{frame}{Developers can plug in at the representation they need}
  \begin{columns}[T]
  \begin{column}{0.33\textwidth}
    \begin{block}{CLIF-native features}
      Direct table-based extraction for models that operate on CLIF relations.
    \end{block}
  \end{column}
  \begin{column}{0.33\textwidth}
    \begin{block}{MEDS event stream}
      Optional event-sequence export for MEDS-native pipelines.
    \end{block}
  \end{column}
  \begin{column}{0.33\textwidth}
    \begin{block}{Wide table}
      YAIB-style matrix for classical ML and one-row-per-prediction workflows.
    \end{block}
  \end{column}
  \end{columns}
  \vfill
  \begin{center}\color{flairblue}\textbf{FLAIR standardizes the task and evaluation contract, not the model architecture.}\end{center}
  {\color{flairmuted}\small Model teams can use their preferred representation as long as outputs meet the prediction-file contract. FLAIR can assist with supported formats; the model owner controls feature engineering and modeling choices.}
  \source{Arnrich et al. 2025; van de Water et al. 2023; Wornow et al. 2023}
\end{frame}

\begin{frame}{The initial catalog covers decisions ICUs actually face}
  \begin{center}
  \footnotesize

  \rowcolors{2}{flairpanel}{white}

  \end{itemize}

\begin{frame}{Sepsis is the stress test}
  \begin{columns}[T]
  \begin{column}{0.56\textwidth}
    \begin{itemize}
      \item Task 4 uses CDC Adult Sepsis Event within 6 hours on an hourly, hospital-wide grid
      \item The prediction window ends strictly before onset: no future treatment signal
      \item Peak scoring collapses each stay to the maximum risk, matching a one-alarm operational interpretation
      \item It is rare, high-stakes, and sensitive to practice-pattern shift
    \end{itemize}
  \end{column}
  \begin{column}{0.40\textwidth}
    \begin{block}{Why this matters}
      If a sepsis model fails across hospitals, the field needs a safe way to find out before deployment, not after.
    \end{block}
  \end{column}
  \end{columns}
  \source{Kamran et al., NEJM AI 2024; Rhee et al. 2019}
\end{frame}

\section{How It Works}

\begin{frame}{Where FLAIR fits in the AI/CDSS lifecycle}
  \begin{center}
  \scriptsize
  \setlength{\tabcolsep}{4pt}
  \rowcolors{2}{flairpanel}{white}
  \begin{tabular}{>{\raggedright\arraybackslash}p{2.0cm}>{\raggedright\arraybackslash}p{4.7cm}>{\raggedright\arraybackslash}p{3.9cm}}
  \toprule
  \textbf{Step} & \textbf{In plain terms} & \textbf{Owner} \\
  \midrule
  Cohort + task & Decide who is included, when prediction happens, and what counts as the outcome & \emphc{FLAIR standard}; community-extensible \\
  Feature engineering & Turn raw clinical history into model inputs using CLIF-native, MEDS, or wide data & Model owner; FLAIR assists with formats \\
  Model development & Train, tune, document, and freeze the algorithm & Model owner \\
  Federated evaluation & Run the frozen artifact locally at each site using the same task and metrics & \emphc{FLAIR protocol}; site-approved \\
  Evidence report & Return comparable, aggregate, cell-suppressed performance summaries & \emphc{FLAIR report standard} \\
  \bottomrule
  \end{tabular}
  \end{center}
  \vfill
  \begin{center}\color{flairblue}\textbf{Encapsulation makes evaluation faster and cleaner: shared task logic, private models, repeatable reports.}\end{center}
\end{frame}

\begin{frame}{The model-to-data loop}
  \begin{center}
    \includegraphics[width=\linewidth,height=0.70\textheight,keepaspectratio]{figure1_flair_overview.png}
  \end{center}

\begin{frame}{Governance is part of the product}
  \begin{columns}[T]
  \begin{column}{0.50\textwidth}
    \begin{itemize}
      \item Benchmark Committee reviews reproducibility, clinical face validity, task fit, and security
      \item Sites retain final approval over what runs locally
      \item Every report includes task hash, package version, split, report mode, n, prevalence, and threshold
    \end{itemize}
  \end{column}
  \begin{column}{0.46\textwidth}
    \begin{block}{Privacy posture}
      No patient-level data leave any site. Counts below 10 are suppressed; sparse subgroups and calibration bins are withheld.
    \end{block}
  \end{column}
  \end{columns}
  \vfill
  \bigclaim{The network is usable because trust, review, and suppression are built in.}
\end{frame}

\begin{frame}{Comparability safeguards make results credible}
  \begin{enumerate}
    \item \emphc{Stitched hospitalizations:} fragmented clinical episodes are joined into one \texttt{hospitalization\_join\_id}; one episode never straddles train/test
    \item \emphc{Strict point-in-time features:} only data at or before \texttt{prediction\_dttm} can be used
    \item \emphc{Pseudo-prospective external splits:} default is oldest 75\% train, newest 25\% test at each site
    \item \emphc{Audit hashes:} sites can prove they ran byte-identical task code and policy metadata
  \end{enumerate}
  \vfill
  {\color{flairmuted}\small MIMIC-CLIF uses a deterministic hash split because de-identified dates do not preserve usable temporal order.}
\end{frame}

\begin{frame}{Evaluation matches the way a task is used}
  \begin{center}
  \scriptsize
  \setlength{\tabcolsep}{5pt}
  \rowcolors{2}{flairpanel}{white}
  \begin{tabular}{>{\raggedright\arraybackslash}p{1.7cm}>{\raggedright\arraybackslash}p{3.2cm}>{\raggedright\arraybackslash}p{6.2cm}}
  \toprule
  \textbf{Mode} & \textbf{Prediction unit} & \textbf{Report bundle} \\
  \midrule
  Episodic (T3, T5) & one prediction per stay & AUROC, AUPRC, ROC/PR curves, calibration, decision curves, subgroup AUROC \\
  Landmark (T1, T2) & one daily prediction while at risk & per-landmark AUC curve plus pooled dynamic AUC, lead-time and skill plots \\
  Peak (T4) & stay collapsed to maximum risk & discrimination and subgroup metrics; calibration/DCA omitted to avoid peak-calibration distortion \\
  \bottomrule
  \end{tabular}
  \end{center}
  \vfill
  \begin{center}\color{flairblue}\textbf{The benchmark measures operational behavior, not just row-level discrimination.}\end{center}
\end{frame}

\begin{frame}{The reference baseline is deliberately shippable}
  \begin{itemize}
    \item One frozen XGBoost model over point-in-time count features, following EHRSHOT/FEMR tradition
    \item CLIF events map to a fixed 211-code Event List Format vocabulary
    \item Counts include only codes strictly before prediction time; units and discharge diagnoses are excluded
    \item Hyperparameters tuned once on MIMIC-CLIF; model and dictionary ship unchanged to sites for inference only
  \end{itemize}
  \vfill
  \begin{center}\color{flairmuted}\small The baseline is not the product. It is the benchmark check that exposes transport behavior.\end{center}
  \source{Wornow et al. 2023; Steinberg et al. 2021}
\end{frame}

\section{Why It Wins}

\begin{frame}{FLAIR's defensible advantages}
  \begin{itemize}
    \item \emphc{Saturation resistance:} evaluation data stay siloed and private, limiting tune-to-test and pretraining-on-test-set risks
    \item \emphc{Evaluation, not training:} aggregate metrics leak less than federated-learning updates; no per-site model object is shared
    \item \emphc{Auditability:} byte-identical code and hashes make labels, exclusions, splits, and cohorts re-inspectable
    \item \emphc{Extensibility:} new tasks can enter through PRs and multi-site governance instead of one-off benchmark forks
    \item \emphc{Adoption flywheel:} every CLIF site increases the value of every future model evaluation
  \end{itemize}
\end{frame}

\begin{frame}{What FLAIR is not claiming}
  \begin{itemize}
    \item It does not certify any model for bedside use
    \item It does not replace prospective silent trials or clinical-impact studies
    \item It does not move patient-level data or expose row-level errors centrally
    \item It currently covers structured EHR binary prediction tasks, not notes, waveforms, imaging, treatment effects, or generative outputs
  \end{itemize}
  \vfill

\begin{frame}{Backup: pooled dynamic AUC for landmark tasks}
  \small
  For a stitched hospitalization $h$ with event onset $T_h$, discharge $D_h$, horizon $\tau$, at landmark time $t$:
  \vskip0.3em
  \begin{columns}[T]
  \begin{column}{0.5\textwidth}
    \setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{4pt}
    Risk set:
    \[ R(t)=\{\,h : T_h>t,\ D_h>t\,\} \]
    Dynamic label:
    \[ Y_h(t)=\mathbf 1\{\,t<T_h\le t+\tau\,\} \]
    Predicted risk:
    \[ p_h(t)=\hat P\!\left(t<T_h\le t+\tau \mid \mathcal H_h(t)\right) \]
  \end{column}
  \begin{column}{0.5\textwidth}
    \setlength{\abovedisplayskip}{2pt}\setlength{\belowdisplayskip}{4pt}
    Tie kernel: $\psi(a,b)=\mathbf 1\{a>b\}+\tfrac12\mathbf 1\{a=b\}$
    \[ \resizebox{\linewidth}{!}{$\displaystyle \widehat{\mathrm{AUC}}(t)=\frac{\sum_{i,j\in R(t)}Y_i(t)\,[1-Y_j(t)]\,\psi\big(p_i(t),p_j(t)\big)}{\big(\sum_i Y_i(t)\big)\big(\sum_j [1-Y_j(t)]\big)}$} \]
    \[ \resizebox{\linewidth}{!}{$\displaystyle \mathrm{AUC}_{\text{pooled}}=\frac{\sum_t \widehat{\mathrm{AUC}}(t)\,w_t}{\sum_t w_t},\ \ w_t=\text{cases}_t\!\cdot\!\text{controls}_t$} \]
  \end{column}
  \end{columns}
  \source{Heagerty \& Zheng 2005; van Houwelingen 2007; Blanche et al. 2013}
\end{frame}
